  채널 상태 관리

  📊 Channel State Management System

  ┌─────────────────────────────────────┐
  │           Channel States             │
  ├─────────────────────────────────────┤
  │ 🟢 ACTIVE    : 실제 통화 중         │
  │ 🟡 STANDBY   : 연결됨, 대기 중       │
  │ 🔴 IDLE      : 연결 해제됨           │
  │ ⚡ JOINING   : 참여 중              │
  │ 🚨 EMERGENCY : 긴급 상황            │
  └─────────────────────────────────────┘

  State Transition Rules:
  IDLE → JOINING (PTT 요청 시)
  JOINING → STANDBY (연결 완료)
  STANDBY → ACTIVE (음성 송신 시)
  ACTIVE → STANDBY (송신 중지)
  STANDBY → IDLE (5분 무활동)

  // SmartChannelManager.kt - 핵심 시스템
  class SmartChannelManager(
      private val pttController: PTTController,
      private val fcmManager: FCMSignalingManager,
      private val scope: CoroutineScope
  ) {
      private var channelState = MutableStateFlow(ChannelState.IDLE)
      private var lastActivity = 0L
      private var idleTimer: Job? = null

      // 🎯 핵심 기능: 필요시에만 연결
      suspend fun requestConnection(trigger: ConnectionTrigger): Result<Unit> {
          return when (trigger) {
              ConnectionTrigger.VOLUME_KEY_PRESSED -> {
                  // 즉시 연결 + 다른 사용자들에게 알림
                  connectAndNotify(priority = Priority.HIGH)
              }
              ConnectionTrigger.INCOMING_PTT_SIGNAL -> {
                  // 빠른 연결 (알림 받음)
                  connectSilently(priority = Priority.NORMAL)
              }
              ConnectionTrigger.EMERGENCY_CALL -> {
                  // 최우선 연결
                  connectAndNotify(priority = Priority.EMERGENCY)
              }
          }
      }

      // 🔄 자동 연결 해제 관리
      private fun startIdleTimer() {
          idleTimer?.cancel()
          idleTimer = scope.launch {
              delay(IDLE_TIMEOUT_MS)
              if (shouldDisconnect()) {
                  disconnectGracefully()
              }
          }
      }
  }

  // FCMSignalingManager.kt - 실시간 PTT 신호 관리
  class FCMSignalingManager(
      private val functions: FirebaseFunctions,
      private val context: Context
  ) {

      // 🚨 PTT 시작 알림 전송
      suspend fun notifyPTTStart(
          channelName: String,
          senderUID: Int,
          regionId: String,
          officeId: String
      ) {
          val payload = mapOf(
              "type" to "PTT_START_SIGNAL",
              "channel" to channelName,
              "senderUID" to senderUID,
              "region" to regionId,
              "office" to officeId,
              "timestamp" to System.currentTimeMillis(),
              "priority" to "high"
          )

          // Cloud Function 호출로 같은 사무실 모든 기기에 알림
          functions.getHttpsCallable("sendPTTSignal")
              .call(payload)
              .await()
      }

      // 📱 FCM 메시지 처리
      fun handleIncomingSignal(message: RemoteMessage) {
          when (message.data["type"]) {
              "PTT_START_SIGNAL" -> {
                  val channel = message.data["channel"]!!
                  val senderUID = message.data["senderUID"]!!.toInt()

                  // 즉시 채널 참여 요청
                  requestAutoJoinChannel(channel, senderUID)
              }

              "EMERGENCY_BROADCAST" -> {
                  // 최우선 긴급 연결
                  requestEmergencyConnection(message.data)
              }
          }
      }
  }

  // Cloud Functions - PTT 신호 중계 서버
  // functions/src/index.ts

  export const sendPTTSignal = functions
    .region('asia-northeast3')
    .https.onCall(async (data, context) => {

      const { region, office, senderUID, channel } = data;

      // 같은 사무실 모든 FCM 토큰 조회
      const tokensSnapshot = await admin.firestore()
        .collection(`regions/${region}/offices/${office}/designated_drivers`)
        .where('fcmToken', '!=', null)
        .get();

      const tokens = tokensSnapshot.docs
        .map(doc => doc.data().fcmToken)
        .filter(token => token); // null 제거

      if (tokens.length === 0) return;

      // 일괄 알림 전송
      const message = {
        data: {
          type: 'PTT_START_SIGNAL',
          channel: channel,
          senderUID: senderUID.toString(),
          autoJoin: 'true'
        },
        android: {
          priority: 'high',
          ttl: 3000 // 3초 후 만료
        }
      };

      await admin.messaging().sendMulticast({
        tokens: tokens,
        ...message
      });

      return { success: true, notifiedDevices: tokens.length };
    });

  // PTTSignalingService.kt - FCM 메시지 처리 강화
  class PTTSignalingService : FirebaseMessagingService() {

      override fun onMessageReceived(message: RemoteMessage) {
          val smartManager = SmartChannelManager.getInstance()

          when (message.data["type"]) {
              "PTT_START_SIGNAL" -> {
                  // 다른 사용자가 PTT 시작 → 자동 참여
                  smartManager.handleIncomingPTTSignal(
                      channelName = message.data["channel"]!!,
                      senderUID = message.data["senderUID"]!!.toInt()
                  )
              }

              "CHANNEL_ACTIVITY" -> {
                  // 채널에 활동이 있음 → 필요시 참여
                  smartManager.handleChannelActivity(message.data)
              }
          }
      }
  }

  // UsageAnalyzer.kt - AI 기반 패턴 학습
  class UsageAnalyzer(private val context: Context) {
      private val prefs = context.getSharedPreferences("usage_patterns", Context.MODE_PRIVATE)

      // 📊 사용 패턴 학습
      fun learnFromUsage(session: PTTSession) {
          val pattern = UserPattern(
              timeOfDay = session.startTime.hour,
              dayOfWeek = session.startTime.dayOfWeek.value,
              duration = session.duration,
              batteryLevel = session.batteryLevelAtStart,
              connectionDelay = session.connectionDelay
          )

          // 패턴 저장 및 분석
          savePattern(pattern)
          updateOptimizationStrategy()
      }

      // 🎯 예측 기반 최적화
      fun predictOptimalStrategy(): OptimizationStrategy {
          val recentPatterns = getRecentPatterns(days = 7)

          return when {
              recentPatterns.averageConnectionDelay < 1000 ->
                  OptimizationStrategy.INSTANT_CONNECTION

              recentPatterns.shortCallsRatio > 0.8 ->
                  OptimizationStrategy.QUICK_DISCONNECT

              recentPatterns.batteryConstrainedUsage ->
                  OptimizationStrategy.POWER_SAVER

              else -> OptimizationStrategy.BALANCED
          }
      }
  }
