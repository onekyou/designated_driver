## 🎯 락인(Lock-in) 시스템 (최우선)

### 락인의 3가지 축

```typescript
1. 데이터 락인 (Data Lock-in) - 50% 가중치
   └─ 고객 데이터가 많을수록 떠나기 어려움
   └─ 고객 수, 활성 고객 비율, 콜 내역

2. 경제적 락인 (Financial Lock-in) - 35% 가중치
   └─ 포인트 부채가 클수록 떠나기 어려움
   └─ 포인트 잔액, VIP/GOLD 고객 비율

3. 네트워크 락인 (Network Lock-in) - 15% 가중치
   └─ 공유콜 연결이 많을수록 떠나기 어려움
   └─ 공유콜 네트워크 수, 공유콜 거래량
```

### 락인 점수 계산 (0-100점)

```typescript
// lib/services/lockInCalculator.ts

export interface LockInScore {
  dataLockIn: number;        // 데이터 락인 (0-100)
  financialLockIn: number;   // 경제적 락인 (0-100)
  networkLockIn: number;     // 네트워크 락인 (0-100)
  totalScore: number;        // 전체 락인 점수 (0-100)
  churnRisk: 'low' | 'medium' | 'high';
  conversionProbability: number; // 유료 전환 확률 (0-100%)
}

/**
 * 데이터 락인 점수 (0-100)
 */
function calculateDataLockIn(office): number {
  let score = 0;

  // 총 고객 수 (최대 50점)
  if (customers >= 200) score += 50;
  else if (customers >= 100) score += 40;
  else if (customers >= 50) score += 30;
  else score += (customers / 50) * 30;

  // 활성 고객 비율 (최대 30점)
  const activeRatio = activeCustomers / customers;
  score += activeRatio * 30;

  // 평균 콜 수 (최대 20점)
  const avgCallsPerCustomer = totalCalls / customers;
  if (avgCallsPerCustomer >= 5) score += 20;
  else score += (avgCallsPerCustomer / 5) * 20;

  return Math.min(score, 100);
}

/**
 * 경제적 락인 점수 (0-100)
 */
function calculateFinancialLockIn(office): number {
  let score = 0;

  // 포인트 잔액 (최대 60점)
  if (pointsBalance >= 5000000) score += 60;      // 500만원
  else if (pointsBalance >= 3000000) score += 50; // 300만원
  else if (pointsBalance >= 1000000) score += 40; // 100만원
  else score += (pointsBalance / 1000000) * 40;

  // VIP/GOLD 고객 비율 (최대 40점)
  const premiumRatio = (vipCount + goldCount) / customers;
  score += premiumRatio * 40;

  return Math.min(score, 100);
}

/**
 * 네트워크 락인 점수 (0-100)
 */
function calculateNetworkLockIn(office): number {
  let score = 0;

  // 공유콜 연결 수 (최대 50점)
  if (connections >= 10) score += 50;
  else if (connections >= 5) score += 40;
  else score += (connections / 5) * 40;

  // 공유콜 거래량 (최대 50점)
  if (sharedCallCount >= 50) score += 50;
  else score += (sharedCallCount / 50) * 50;

  return Math.min(score, 100);
}

/**
 * 전체 락인 점수 (가중 평균)
 */
function calculateTotalLockIn(office): number {
  return (
    dataLockIn * 0.5 +
    financialLockIn * 0.35 +
    networkLockIn * 0.15
  );
}
```

## 🔄 적응형 유료 전환 시스템

### 고정 기간의 문제점

```
❌ 모든 사무실에 3개월 무료:

  홍천 대리운전 (우수):
  - 1개월 만에 락인 85점 달성
  - 그런데 3개월까지 무료?
  - → 2개월 수익 손실

  속초 대리운전 (부진):
  - 3개월 동안 락인 25점
  - 3개월 후 유료 전환?
  - → 바로 이탈
```

### 적응형 전환 (락인 점수 기반)

```
✅ 개별 맞춤 전환:

  홍천 대리운전:
  - 1개월 만에 락인 85점
  - 시스템: "유료 전환 가능!"
  - → 즉시 유료 전환 제안
  - 전환 성공률: 95%+

  속초 대리운전:
  - 3개월 동안 락인 25점
  - 시스템: "아직 준비 안 됨"
  - → 계속 무료 + 락인 강화 지원
  - 6개월 후 락인 70점 달성 → 그때 유료 전환
```

interface ConversionCriteria {
  // 기본 조건

  hasEnoughCustomers: 50,          // 고객 50명 이상
  hasPointsDebt: 1000000,          // 포인트 100만원 이상
  minActiveUsageDays: 14,          // 14일 이상 활발한 사용
  minSharedCallConnections: 3,     // 공유콜 3개 이상 연결

  // 제약 조건
  minTrialDays: 21,                // 최소 3주는 무료
  maxTrialDays: 180,               // 최대 6개월 무료
}

@Composable
fun WithdrawalRequestDialog(
    onDismiss: () -> Unit,
    onSubmit: (amount: Int, bankInfo: BankInfo) -> Unit
) {
    var amount by remember { mutableStateOf("") }
    var bankName by remember { mutableStateOf("") }
    var accountNumber by remember { mutableStateOf("") }
    var accountHolder by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("환전 신청") },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                Text("현재 잔액: ${formatCurrency(pointBalance)}")

                OutlinedTextField(
                    value = amount,
                    onValueChange = { amount = it },
                    label = { Text("환전 금액") },
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                    suffix = { Text("원") }
                )

                if (amount.toIntOrNull() ?: 0 < 100000) {
                    Text(
                        "최소 환전 금액은 10만원입니다",
                        color = Color.Red,
                        fontSize = 12.sp
                    )
                }

                OutlinedTextField(
                    value = bankName,
                    onValueChange = { bankName = it },
                    label = { Text("은행명") }
                )

                OutlinedTextField(
                    value = accountNumber,
                    onValueChange = { accountNumber = it },
                    label = { Text("계좌번호") },
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number)
                )

                OutlinedTextField(
                    value = accountHolder,
                    onValueChange = { accountHolder = it },
                    label = { Text("예금주") }
                )
            }
        },
        confirmButton = {
            Button(
                onClick = {
                    val amountInt = amount.toIntOrNull() ?: 0
                    if (amountInt >= 100000) {
                        onSubmit(
                            amountInt,
                            BankInfo(bankName, accountNumber, accountHolder)
                        )
                        onDismiss()
                    }
                },
                enabled = (amount.toIntOrNull() ?: 0) >= 100000 &&
                          bankName.isNotBlank() &&
                          accountNumber.isNotBlank() &&
                          accountHolder.isNotBlank()
            ) {
                Text("신청")
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text("취소")
            }
        }
    )
}

// functions/src/withdrawal.ts

import * as functions from "firebase-functions";
import * as admin from "firebase-admin";

const db = admin.firestore();

/**
 * 환전 신청 시: 잔액 검증 및 알림
 */
export const onWithdrawalRequested = functions.firestore
  .document("regions/{regionId}/withdrawalRequests/{requestId}")
  .onCreate(async (snapshot, context) => {
    const request = snapshot.data();
    const { regionId } = context.params;

    // 현재 잔액 조회
    const pointsRef = db.doc(
      `regions/${regionId}/offices/${request.officeId}/points/points`
    );
    const pointsDoc = await pointsRef.get();
    const currentBalance = pointsDoc.data()?.currentPoints || 0;

    // 잔액 부족 시 자동 거부
    if (currentBalance < request.amount) {
      await snapshot.ref.update({
        status: 'rejected',
        rejectedReason: '잔액 부족',
        reviewedAt: admin.firestore.FieldValue.serverTimestamp()
      });

      // 사무실에 알림
      await notifyOfficeAdmin(request.officeId, {
        title: '환전 신청 거부',
        body: `잔액 부족으로 환전 신청이 자동 거부되었습니다.`
      });

      return;
    }

    // 총관리자에게 알림
    await notifyHeadManagers({
      title: '환전 신청',
      body: `${request.officeName}에서 ${formatCurrency(request.amount)} 환전 신청`,
      data: {
        type: 'WITHDRAWAL_REQUEST',
        requestId: snapshot.id
      }
    });
  });

/**
 * 환전 승인 시: 사무실에 알림
 */
export const onWithdrawalApproved = functions.firestore
  .document("regions/{regionId}/withdrawalRequests/{requestId}")
  .onUpdate(async (change, context) => {
    const before = change.before.data();
    const after = change.after.data();

    // pending → approved 변경 감지
    if (before.status === 'pending' && after.status === 'approved') {
      // 사무실 관리자에게 알림
      await notifyOfficeAdmin(after.officeId, {
        title: '환전 승인',
        body: `${formatCurrency(after.amount)} 환전 신청이 승인되었습니다. 곧 송금됩니다.`,
        data: {
          type: 'WITHDRAWAL_APPROVED',
          requestId: context.params.requestId
        }
      });
    }
  });

/**
 * 송금 완료 처리: 포인트 실제 차감 + 알림
 */
export const onWithdrawalCompleted = functions.firestore
  .document("regions/{regionId}/withdrawalRequests/{requestId}")
  .onUpdate(async (change, context) => {
    const before = change.before.data();
    const after = change.after.data();
    const { regionId, requestId } = context.params;

    // approved → completed 변경 감지
    if (before.status === 'approved' && after.status === 'completed') {

      // 트랜잭션으로 포인트 차감
      await db.runTransaction(async (transaction) => {
        const pointsRef = db.doc(
          `regions/${regionId}/offices/${after.officeId}/points/points`
        );

        const pointsDoc = await transaction.get(pointsRef);
        const currentPoints = pointsDoc.data()?.currentPoints || 0;

        // 포인트 차감
        transaction.update(pointsRef, {
          currentPoints: currentPoints - after.amount,
          totalWithdrawn: (pointsDoc.data()?.totalWithdrawn || 0) + after.amount,
          lastWithdrawalAt: admin.firestore.FieldValue.serverTimestamp()
        });

        // 거래 내역 저장
        const txRef = db.collection(
          `regions/${regionId}/offices/${after.officeId}/point_transactions`
        ).doc();

        transaction.set(txRef, {
          type: 'WITHDRAWAL',
          amount: -after.amount,
          description: '공유콜 포인트 환전',
          withdrawalRequestId: requestId,
          balanceAfter: currentPoints - after.amount,
          createdAt: admin.firestore.FieldValue.serverTimestamp()
        });

        // 환전 내역 저장
        const historyRef = db.collection(
          `regions/${regionId}/offices/${after.officeId}/withdrawalHistory`
        ).doc();

        transaction.set(historyRef, {
          requestId: requestId,
          amount: after.amount,
          actualAmount: after.actualAmount || after.amount,
          fee: after.fee || 0,
          status: 'completed',
          completedAt: admin.firestore.FieldValue.serverTimestamp(),
          bankInfo: after.bankInfo
        });
      });

      // 사무실에 완료 알림
      await notifyOfficeAdmin(after.officeId, {
        title: '송금 완료',
        body: `${formatCurrency(after.amount)}이 입금되었습니다.`,
        data: {
          type: 'WITHDRAWAL_COMPLETED',
          requestId: requestId,
          amount: after.amount.toString()
        }
      });
    }
  });

/**
 * 헬퍼 함수들
 */
async function notifyHeadManagers(notification: any) {
  const headManagersSnapshot = await db
    .collection('admins')
    .where('role', 'in', ['HEAD_MANAGER', 'SUPER_ADMIN'])
    .get();

  const tokens = headManagersSnapshot.docs
    .map(doc => doc.data().fcmToken)
    .filter(token => token);

  if (tokens.length > 0) {
    await admin.messaging().sendMulticast({
      tokens,
      notification: {
        title: notification.title,
        body: notification.body
      },
      data: notification.data || {}
    });
  }
}

async function notifyOfficeAdmin(officeId: string, notification: any) {
  const adminSnapshot = await db
    .collection('admins')
    .where('officeId', '==', officeId)
    .limit(1)
    .get();

  if (!adminSnapshot.empty) {
    const adminData = adminSnapshot.docs[0].data();
    if (adminData.fcmToken) {
      await admin.messaging().send({
        token: adminData.fcmToken,
        notification: {
          title: notification.title,
          body: notification.body
        },
        data: notification.data || {}
      });
    }
  }
}

function formatCurrency(amount: number): string {
  return `${amount.toLocaleString('ko-KR')}원`;
}
