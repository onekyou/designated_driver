## 전체 개요

### 목표
- **모니터링**: 모든 앱의 에러를 중앙에서 실시간으로 감지 및 추적
- **복구**: 자동 복구 가능한 에러는 즉시 처리, 불가능한 것은 관리자에게 알림
- **분석**: 에러 패턴 분석으로 사전 예방 및 시스템 개선

### 대상 앱
1. Call Detector (콜 감지 앱)
2. Call Manager (관리자 앱)
3. Driver App (기사 앱)
4. Customer App (고객 앱)
5. Cloud Functions (서버)
6. Head Manager Web (총관리자 웹)

### 핵심 전략
```
┌─────────────────────────────────────┐
│  Firebase Crashlytics               │
│  (앱 크래시 & 기술적 오류)           │
│  - 자동 수집                         │
│  - Firebase Console에서 확인         │
└─────────────────────────────────────┘
              +
┌─────────────────────────────────────┐
│  Firestore 커스텀 로그              │
│  (비즈니스 로직 오류)                │
│  - 선별적 기록                       │
│  - 총관리자 웹에서 실시간 확인       │
└─────────────────────────────────────┘
              +
┌─────────────────────────────────────┐
│  재시도 & 자동 복구 로직            │
│  (각 앱/Functions에 내장)            │
│  - 네트워크 오류 자동 재시도         │
│  - 일시적 오류 자동 복구             │
└─────────────────────────────────────┘
```

```kotlin
// shared/ErrorLogger.kt
object ErrorLogger {
    private val firestore = FirebaseFirestore.getInstance()
    private val auth = FirebaseAuth.getInstance()

    data class ErrorContext(
        val appType: String,
        val category: String,
        val severity: String,
        val errorCode: String,
        val errorMessage: String,
        val metadata: Map<String, Any> = emptyMap(),
        val stackTrace: String? = null
    )

    suspend fun log(context: ErrorContext) {
        try {
            val errorLog = hashMapOf(
                "timestamp" to FieldValue.serverTimestamp(),
                "appType" to context.appType,
                "category" to context.category,
                "severity" to context.severity,
                "errorCode" to context.errorCode,
                "errorMessage" to context.errorMessage,
                "metadata" to context.metadata,
                "stackTrace" to context.stackTrace,
                "userId" to (auth.currentUser?.uid ?: "anonymous"),
                "regionId" to context.metadata["regionId"],
                "officeId" to context.metadata["officeId"],
                "autoRecoveryAttempted" to false,
                "autoRecoverySuccess" to false,
                "retryCount" to 0,
                "status" to "open",
                "acknowledgedBy" to null,
                "resolvedBy" to null
            )

            firestore.collection("errorLogs")
                .add(errorLog)
                .await()

        } catch (e: Exception) {
            // 로깅 실패 시 로컬에 저장 (나중에 재시도)
            Log.e("ErrorLogger", "Failed to log error", e)
        }
    }

    // 편의 함수들
    suspend fun logCritical(
        category: String,
        errorCode: String,
        message: String,
        metadata: Map<String, Any> = emptyMap()
    ) {
        log(ErrorContext(
            appType = "call_manager", // 각 앱마다 다르게
            category = category,
            severity = "critical",
            errorCode = errorCode,
            errorMessage = message,
            metadata = metadata
        ))
    }

    suspend fun logError(
        category: String,
        errorCode: String,
        message: String,
        metadata: Map<String, Any> = emptyMap()
    ) {
        log(ErrorContext(
            appType = "call_manager",
            category = category,
            severity = "error",
            errorCode = errorCode,
            errorMessage = message,
            metadata = metadata
        ))
    }

    suspend fun logWarning(
        category: String,
        errorCode: String,
        message: String,
        metadata: Map<String, Any> = emptyMap()
    ) {
        log(ErrorContext(
            appType = "call_manager",
            category = category,
            severity = "warning",
            errorCode = errorCode,
            errorMessage = message,
            metadata = metadata
        ))
    }
}
```

#### 2.2 Functions 에러 로깅

```typescript
// functions/src/utils/errorLogger.ts
import * as admin from 'admin';
import * as logger from 'firebase-functions/logger';

interface ErrorContext {
  category: string;
  severity: 'critical' | 'error' | 'warning' | 'info';
  errorCode: string;
  errorMessage: string;
  metadata?: Record<string, any>;
  stackTrace?: string;
}

export async function logError(context: ErrorContext): Promise<void> {
  try {
    await admin.firestore().collection('errorLogs').add({
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
      appType: 'functions',
      category: context.category,
      severity: context.severity,
      errorCode: context.errorCode,
      errorMessage: context.errorMessage,
      metadata: context.metadata || {},
      stackTrace: context.stackTrace || null,
      userId: null,
      regionId: context.metadata?.regionId || null,
      officeId: context.metadata?.officeId || null,
      autoRecoveryAttempted: false,
      autoRecoverySuccess: false,
      retryCount: 0,
      status: 'open',
      acknowledgedBy: null,
      resolvedBy: null,
    });

    // 로컬 로그에도 기록
    logger.error(context.errorMessage, context.metadata);

  } catch (error) {
    // 로깅 실패 시 최소한 로컬에는 남김
    logger.error('Failed to log error to Firestore', error);
  }
}

export const logCritical = (
  category: string,
  errorCode: string,
  message: string,
  metadata?: Record<string, any>
) => logError({ category, severity: 'critical', errorCode, errorMessage: message, metadata });

export const logErrorLevel = (
  category: string,
  errorCode: string,
  message: string,
  metadata?: Record<string, any>
) => logError({ category, severity: 'error', errorCode, errorMessage: message, metadata });
```

#### 3.2 Hook 구현

```typescript
// head_manager_web/lib/hooks/useErrorLogs.ts
'use client';

import { useEffect, useState } from 'react';
import { db } from '../firebase';
import {
  collection,
  query,
  where,
  orderBy,
  limit,
  onSnapshot,
  doc,
  updateDoc,
  serverTimestamp,
  Timestamp,
} from 'firebase/firestore';
import { ErrorLog, ErrorStats } from '../types/errorLog';
import { useAuth } from './useAuth';

export function useErrorLogs() {
  const { user } = useAuth();
  const [logs, setLogs] = useState<ErrorLog[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // 실시간 구독
  useEffect(() => {
    if (!user) return;

    const q = query(
      collection(db, 'errorLogs'),
      orderBy('timestamp', 'desc'),
      limit(100)
    );

    const unsubscribe = onSnapshot(
      q,
      (snapshot) => {
        const errorLogs: ErrorLog[] = [];

        snapshot.forEach((doc) => {
          const data = doc.data();
          errorLogs.push({
            id: doc.id,
            timestamp: data.timestamp?.toDate() || new Date(),
            appType: data.appType,
            category: data.category,
            severity: data.severity,
            errorCode: data.errorCode,
            errorMessage: data.errorMessage,
            stackTrace: data.stackTrace,
            userId: data.userId,
            regionId: data.regionId,
            officeId: data.officeId,
            metadata: data.metadata || {},
            autoRecoveryAttempted: data.autoRecoveryAttempted || false,
            autoRecoverySuccess: data.autoRecoverySuccess || false,
            retryCount: data.retryCount || 0,
            status: data.status || 'open',
            acknowledgedBy: data.acknowledgedBy,
            acknowledgedAt: data.acknowledgedAt?.toDate(),
            resolvedBy: data.resolvedBy,
            resolvedAt: data.resolvedAt?.toDate(),
            resolutionNote: data.resolutionNote,
            relatedErrorIds: data.relatedErrorIds || [],
            affectedUsers: data.affectedUsers || 1,
          });
        });

        setLogs(errorLogs);
        setLoading(false);
      },
      (err) => {
        console.error('에러 로그 조회 실패:', err);
        setError('에러 로그를 불러오는데 실패했습니다.');
        setLoading(false);
      }
    );

    return () => unsubscribe();
  }, [user]);

  // 에러 확인 (acknowledge)
  const acknowledgeError = async (errorId: string) => {
    if (!user) return;

    try {
      const errorRef = doc(db, 'errorLogs', errorId);
      await updateDoc(errorRef, {
        status: 'acknowledged',
        acknowledgedBy: user.uid,
        acknowledgedAt: serverTimestamp(),
      });
    } catch (err) {
      console.error('에러 확인 실패:', err);
      throw err;
    }
  };

  // 에러 해결
  const resolveError = async (errorId: string, resolutionNote?: string) => {
    if (!user) return;

    try {
      const errorRef = doc(db, 'errorLogs', errorId);
      await updateDoc(errorRef, {
        status: 'resolved',
        resolvedBy: user.uid,
        resolvedAt: serverTimestamp(),
        resolutionNote: resolutionNote || null,
      });
    } catch (err) {
      console.error('에러 해결 실패:', err);
      throw err;
    }
  };

  // 에러 무시
  const ignoreError = async (errorId: string) => {
    if (!user) return;

    try {
      const errorRef = doc(db, 'errorLogs', errorId);
      await updateDoc(errorRef, {
        status: 'ignored',
      });
    } catch (err) {
      console.error('에러 무시 실패:', err);
      throw err;
    }
  };

  // 필터링된 로그들
  const criticalLogs = logs.filter((log) => log.severity === 'critical');
  const errorLogs = logs.filter((log) => log.severity === 'error');
  const openLogs = logs.filter((log) => log.status === 'open');
  const unresolvedLogs = logs.filter((log) => log.status !== 'resolved' && log.status !== 'ignored');

  // 통계
  const stats: ErrorStats = {
    critical: criticalLogs.length,
    error: errorLogs.length,
    warning: logs.filter((log) => log.severity === 'warning').length,
    info: logs.filter((log) => log.severity === 'info').length,
    total: logs.length,

    byApp: logs.reduce((acc, log) => {
      if (!acc[log.appType]) {
        acc[log.appType] = { critical: 0, error: 0, warning: 0, info: 0 };
      }
      acc[log.appType][log.severity]++;
      return acc;
    }, {} as Record<string, any>),

    byCategory: logs.reduce((acc, log) => {
      acc[log.category] = (acc[log.category] || 0) + 1;
      return acc;
    }, {} as Record<string, number>),

    autoRecovered: logs.filter((log) => log.autoRecoverySuccess).length,
    unresolved: unresolvedLogs.length,
  };

  return {
    logs,
    loading,
    error,
    stats,
    criticalLogs,
    errorLogs,
    openLogs,
    unresolvedLogs,
    acknowledgeError,
    resolveError,
    ignoreError,
  };
}
```

```kotlin
// shared/RetryUtils.kt
object RetryUtils {
    suspend fun <T> retryOnFailure(
        maxRetries: Int = 3,
        initialDelay: Long = 1000, // 1초
        maxDelay: Long = 10000, // 10초
        factor: Double = 2.0, // 지수 백오프
        block: suspend () -> T
    ): T {
        var currentDelay = initialDelay
        var lastException: Exception? = null

        repeat(maxRetries) { attempt ->
            try {
                return block()
            } catch (e: Exception) {
                lastException = e

                // 재시도 불가능한 오류는 즉시 던짐
                if (!isRetryableError(e)) {
                    throw e
                }

                // 마지막 시도가 아니면 대기 후 재시도
                if (attempt < maxRetries - 1) {
                    delay(currentDelay)
                    currentDelay = (currentDelay * factor).toLong().coerceAtMost(maxDelay)
                }
            }
        }

        throw lastException ?: Exception("Retry failed")
    }

    private fun isRetryableError(e: Exception): Boolean {
        return when (e) {
            is java.net.SocketTimeoutException,
            is java.net.UnknownHostException,
            is java.io.IOException,
            is com.google.firebase.FirebaseException -> true
            else -> false
        }
    }
}
```

```typescript
// functions/src/utils/retryUtils.ts
export async function retryWithBackoff<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3,
  initialDelay: number = 1000,
  maxDelay: number = 10000,
  factor: number = 2
): Promise<T> {
  let currentDelay = initialDelay;
  let lastError: Error;

  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error as Error;

      // 재시도 불가능한 오류는 즉시 던짐
      if (!isRetryableError(error)) {
        throw error;
      }

      // 마지막 시도가 아니면 대기
      if (attempt < maxRetries - 1) {
        await new Promise(resolve => setTimeout(resolve, currentDelay));
        currentDelay = Math.min(currentDelay * factor, maxDelay);
      }
    }
  }

  throw lastError!;
}

function isRetryableError(error: any): boolean {
  // Firestore 동시성 오류
  if (error.code === 'aborted' || error.code === 'unavailable') {
    return true;
  }

  // 네트워크 오류
  if (error.code === 'ECONNRESET' || error.code === 'ETIMEDOUT') {
    return true;
  }

  return false;
}
```

```typescript
import { getToken, onMessage } from 'firebase/messaging';
import { doc, updateDoc } from 'firebase/firestore';
import { db } from '../firebase';
import { useAuth } from './useAuth';

export function useFCM() {
  const { user } = useAuth();

  useEffect(() => {
    if (!user || typeof window === 'undefined') return;

    const registerFCM = async () => {
      try {
        // 알림 권한 요청
        const permission = await Notification.requestPermission();
        if (permission !== 'granted') {
          console.log('알림 권한이 거부되었습니다');
          return;
        }

        // FCM 토큰 가져오기
        const token = await getToken(messaging, {
          vapidKey: process.env.NEXT_PUBLIC_FIREBASE_VAPID_KEY,
        });

        // Firestore에 토큰 저장
        await updateDoc(doc(db, 'admins', user.uid), {
          fcmToken: token,
          fcmTokenUpdatedAt: new Date(),
        });

        console.log('FCM 토큰 등록 완료');

        // 포그라운드 메시지 수신
        onMessage(messaging, (payload) => {
          console.log('메시지 수신:', payload);

          // 브라우저 알림 표시
          if (payload.notification) {
            new Notification(payload.notification.title || '알림', {
              body: payload.notification.body,
              icon: '/logo.png',
            });
          }
        });

      } catch (error) {
        console.error('FCM 등록 실패:', error);
      }
    };

    registerFCM();
  }, [user]);
}
```

#### 6.1 일일 통계 집계 Function

```typescript
// functions/src/index.ts
export const aggregateErrorStats = onSchedule(
  {
    schedule: '0 1 * * *', // 매일 새벽 1시
    timeZone: 'Asia/Seoul',
    region: 'asia-northeast3',
  },
  async () => {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 1);
    yesterday.setHours(0, 0, 0, 0);

    const today = new Date(yesterday);
    today.setDate(today.getDate() + 1);

    // 어제 발생한 에러 조회
    const errorsSnapshot = await admin.firestore()
      .collection('errorLogs')
      .where('timestamp', '>=', yesterday)
      .where('timestamp', '<', today)
      .get();

    // 통계 계산
    const stats = {
      date: yesterday.toISOString().split('T')[0],
      byApp: {},
      byCategory: {},
      byOffice: {},
      bySeverity: { critical: 0, error: 0, warning: 0, info: 0 },
      autoRecovered: 0,
      manualResolved: 0,
      unresolved: 0,
      totalErrors: errorsSnapshot.size,
      updatedAt: admin.firestore.FieldValue.serverTimestamp(),
    };

    errorsSnapshot.forEach((doc) => {
      const data = doc.data();

      // 앱별
      if (!stats.byApp[data.appType]) {
        stats.byApp[data.appType] = { critical: 0, error: 0, warning: 0, info: 0 };
      }
      stats.byApp[data.appType][data.severity]++;

      // 카테고리별
      stats.byCategory[data.category] = (stats.byCategory[data.category] || 0) + 1;

      // 사무실별
      if (data.officeId) {
        const key = `${data.regionId}/${data.officeId}`;
        stats.byOffice[key] = (stats.byOffice[key] || 0) + 1;
      }

      // 심각도별
      stats.bySeverity[data.severity]++;

      // 복구 상태별
      if (data.autoRecoverySuccess) {
        stats.autoRecovered++;
      } else if (data.status === 'resolved') {
        stats.manualResolved++;
      } else if (data.status === 'open' || data.status === 'acknowledged') {
        stats.unresolved++;
      }
    });

    // Firestore에 저장
    await admin.firestore()
      .collection('errorStats')
      .doc('daily')
      .collection('dates')
      .doc(stats.date)
      .set(stats);

    logger.info(`일일 에러 통계 집계 완료: ${stats.date}`);
  }
);
```

```typescript
// head_manager_web/app/errors/stats/page.tsx
'use client';

import { useEffect, useState } from 'react';
import { DashboardLayout } from '@/components/layout/DashboardLayout';
import { collection, query, orderBy, limit, getDocs } from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { LineChart, Line, BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend } from 'recharts';

export default function ErrorStatsPage() {
  const [stats, setStats] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchStats();
  }, []);

  const fetchStats = async () => {
    const q = query(
      collection(db, 'errorStats/daily/dates'),
      orderBy('date', 'desc'),
      limit(30)
    );

    const snapshot = await getDocs(q);
    const data = snapshot.docs.map(doc => doc.data()).reverse();
    setStats(data);
    setLoading(false);
  };

  if (loading) {
    return <DashboardLayout><div>로딩 중...</div></DashboardLayout>;
  }

  return (
    <DashboardLayout>
      <div className="space-y-6">
        <h1 className="text-3xl font-bold">에러 통계</h1>

        {/* 일일 에러 트렌드 */}
        <div className="bg-white p-6 rounded-lg shadow">
          <h2 className="text-xl font-semibold mb-4">일일 에러 발생 추이</h2>
          <LineChart width={800} height={300} data={stats}>
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis dataKey="date" />
            <YAxis />
            <Tooltip />
            <Legend />
            <Line type="monotone" dataKey="totalErrors" stroke="#8884d8" name="전체" />
            <Line type="monotone" dataKey="bySeverity.critical" stroke="#ef4444" name="치명적" />

        <div className="bg-white p-6 rounded-lg shadow">
          <h2 className="text-xl font-semibold mb-4">앱별 에러 발생</h2>

            <YAxis />
            <Tooltip />
            <Legend />

            <Bar dataKey="byApp.customer_app.error" fill="#ffc658" name="고객 앱" />
```

   - 구현 후: 자동 감지 → 즉시 확인 → 해결 (평균 10분)

1. **서비스 신뢰도 향상**
   - 문제 발생 전 예방 가능

2. **운영 효율성**
   - 능동적 모니터링 (vs 수동적 확인)
   - 데이터 기반 의사결정

3. **확장성 확보**
   - 사무실 10개 → 100개 되어도 동일한 방식으로 관리
