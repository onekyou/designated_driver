## 🔴 우선순위 1: 초기 비프음 문제 해결

### 문제 분석
```
로그 패턴:
✅ 정상: "PTT start beep played (real sound)"
❌ 실패: "PTT start beep played (system fallback) - initialized: true, soundId: 0"
```

**핵심 문제**: `soundId: 0` → SoundPool.load()가 완료되지 않음

### 해결 방안

#### 방안 1: SoundPool 로딩 완료 대기
```kotlin
class BeepSoundManager(private val context: Context) {
    private var soundPool: SoundPool? = null
    private var startSoundId: Int = 0
    private var isLoadComplete = false
    
    init {
        initialize()
    }
    
    private fun initialize() {
        try {
            soundPool = SoundPool.Builder()
                .setMaxStreams(2)
                .setAudioAttributes(audioAttributes)
                .build()
            
            // 로딩 완료 리스너 추가
            soundPool?.setOnLoadCompleteListener { soundPool, sampleId, status ->
                if (status == 0) { // 성공
                    isLoadComplete = true
                    Log.d(TAG, "Sound loading completed - sampleId: $sampleId")
                } else {
                    Log.e(TAG, "Sound loading failed - status: $status")
                }
            }
            
            startSoundId = soundPool?.load(context, R.raw.ptt_beep, 1) ?: 0
            Log.d(TAG, "Sound loading initiated - soundId: $startSoundId")
            
        } catch (e: Exception) {
            Log.e(TAG, "Failed to initialize BeepSoundManager", e)
        }
    }
    
    fun playStartBeep() {
        try {
            if (isLoadComplete && startSoundId != 0) {
                soundPool?.play(startSoundId, 1.0f, 1.0f, 0, 0, 1.0f)
                Log.d(TAG, "PTT start beep played (real sound)")
            } else {
                playSystemBeep()
                Log.d(TAG, "PTT start beep played (system fallback) - loadComplete: $isLoadComplete, soundId: $startSoundId")
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to play start beep", e)
            playSystemBeep()
        }
    }
}
```

#### 방안 2: MediaPlayer 대체 방안

```kotlin
class BeepSoundManagerV2(private val context: Context) {
    private var mediaPlayer: MediaPlayer? = null
    
    init {
        initialize()
    }
    
    private fun initialize() {
        try {
            mediaPlayer = MediaPlayer.create(context, R.raw.ptt_beep)?.apply {
                setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_MEDIA)
                        .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                        .build()
                )
                prepareAsync()
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to initialize MediaPlayer", e)
        }
    }
    
    fun playStartBeep() {
        try {
            mediaPlayer?.let { player ->
                if (player.isPlaying) {
                    player.seekTo(0) // 처음부터 재생
                } else {
                    player.start()
                }
                Log.d(TAG, "PTT start beep played (MediaPlayer)")
            } ?: playSystemBeep()
        } catch (e: Exception) {
            Log.e(TAG, "Failed to play beep with MediaPlayer", e)
            playSystemBeep()
        }
    }
}
```

## 🔴 우선순위 2: 볼륨키 지속 누름 문제 해결

### 문제 분석
```
현재 로그 패턴:
D PTTAccessibilityService: Volume key pressed - starting PTT
D PTTAccessibilityService: Volume key released - stopping PTT  # ← 너무 빨리 발생
```

### 해결 방안

#### 방안 1: RepeatCount 기반 처리
```kotlin
private fun handleVolumeKeyEvent(event: KeyEvent): Boolean {
    when (event.action) {
        KeyEvent.ACTION_DOWN -> {
            // 첫 번째 DOWN 이벤트만 처리
            if (event.repeatCount == 0) {
                if (!isTransmitting && !isProcessing) {
                    Log.d(TAG, "Volume key pressed - starting PTT (repeatCount: ${event.repeatCount})")
                    isProcessing = true
                    startPTT()
                    isTransmitting = true
                    isProcessing = false
                }
            } else {
                Log.d(TAG, "Volume key repeat ignored (repeatCount: ${event.repeatCount})")
            }
        }
        
        KeyEvent.ACTION_UP -> {
            // UP 이벤트만 처리
            if (isTransmitting && !isProcessing) {
                Log.d(TAG, "Volume key released - stopping PTT")
                isProcessing = true
                stopPTT()
                isTransmitting = false
                isProcessing = false
            }
        }
    }
    return true
}
```

#### 방안 2: Timer 기반 지연 해제
```kotlin
class PTTAccessibilityService : AccessibilityService() {
    private var releaseTimer: Timer? = null
    private val RELEASE_DELAY = 300L // 300ms 지연
    
    private fun handleVolumeKeyEvent(event: KeyEvent): Boolean {
        when (event.action) {
            KeyEvent.ACTION_DOWN -> {
                // 기존 타이머 취소
                releaseTimer?.cancel()
                releaseTimer = null
                
                if (!isTransmitting && !isProcessing) {
                    Log.d(TAG, "Volume key pressed - starting PTT")
                    startPTT()
                    isTransmitting = true
                }
            }
            
            KeyEvent.ACTION_UP -> {
                // 지연된 해제 타이머 시작
                releaseTimer = Timer().apply {
                    schedule(object : TimerTask() {
                        override fun run() {
                            if (isTransmitting) {
                                Log.d(TAG, "Volume key released (delayed) - stopping PTT")
                                stopPTT()
                                isTransmitting = false
                            }
                        }
                    }, RELEASE_DELAY)
                }
            }
        }
        return true
    }
}
```

## 🟡 우선순위 3: 상태 전환 최적화

### 해결 방안
```kotlin
class PTTStateManager {
    private val _state = MutableStateFlow<PTTState>(PTTState.Disconnected)
    val state: StateFlow<PTTState> = _state.asStateFlow()
    
    fun updateState(newState: PTTState) {
        val currentState = _state.value
        
        // 유효한 상태 전환만 허용
        val isValidTransition = when (currentState) {
            is PTTState.Disconnected -> newState is PTTState.Connecting
            is PTTState.Connecting -> newState is PTTState.Connected || newState is PTTState.Error
            is PTTState.Connected -> newState is PTTState.Transmitting || newState is PTTState.Disconnected
            is PTTState.Transmitting -> newState is PTTState.Connected || newState is PTTState.Disconnected
            is PTTState.Error -> true // 에러에서는 모든 상태로 전환 가능
            else -> false
        }
        
        if (isValidTransition) {
            _state.value = newState
            Log.d(TAG, "State transition: $currentState → $newState")
        } else {
            Log.w(TAG, "Invalid state transition: $currentState → $newState")
        }
    }
}
```
