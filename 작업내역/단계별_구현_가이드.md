### 1.7 로그인 페이지 생성

```tsx
// app/login/page.tsx
'use client';

import { useState } from 'react';
import { signInWithEmailAndPassword } from 'firebase/auth';
import { auth } from '@/lib/firebase';
import { useRouter } from 'next/navigation';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';

export default function LoginPage() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const router = useRouter();

  const handleLogin = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setError('');

    try {
      await signInWithEmailAndPassword(auth, email, password);
      router.push('/dashboard');
    } catch (err: any) {
      setError('로그인 실패: ' + err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="flex min-h-screen items-center justify-center bg-gray-100">
      <Card className="w-[400px]">
        <CardHeader>
          <CardTitle className="text-2xl text-center">
            총관리자 로그인
          </CardTitle>
        </CardHeader>
        <CardContent>
          <form onSubmit={handleLogin} className="space-y-4">
            <div>
              <Input
                type="email"
                placeholder="이메일"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                required
              />
            </div>
            <div>
              <Input
                type="password"
                placeholder="비밀번호"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                required
              />
            </div>
            {error && (
              <p className="text-sm text-red-600">{error}</p>
            )}
            <Button
              type="submit"
              className="w-full"
              disabled={loading}
            >
              {loading ? '로그인 중...' : '로그인'}
            </Button>
          </form>
        </CardContent>
      </Card>
    </div>
  );
}
```

```tsx
// app/dashboard/page.tsx
'use client';

import { useEffect, useState } from 'react';
import { auth } from '@/lib/firebase';
import { useRouter } from 'next/navigation';
import { Button } from '@/components/ui/button';
import { signOut } from 'firebase/auth';

export default function DashboardPage() {
```

### 2.1 Firestore 권한 설정

```javascript
// firestore.rules에 추가

// 총관리자 체크 함수
function isHeadManager() {
  let adminDoc = get(/databases/$(database)/documents/admins/$(request.auth.uid));
  return request.auth != null &&
         adminDoc.data.role in ['HEAD_MANAGER', 'SUPER_ADMIN'];
}

// 사무실 데이터 읽기
match /regions/{regionId}/offices/{officeId} {
  allow read: if isHeadManager();
}

// 환전 요청 관리
match /regions/{regionId}/withdrawalRequests/{requestId} {
  allow read, update: if isHeadManager();
}
```

### 2.3 권한 체크 Hook 생성

```typescript
// lib/hooks/useAuth.ts
'use client';

import { useEffect, useState } from 'react';
import { auth, db } from '@/lib/firebase';
import { doc, getDoc } from 'firebase/firestore';
import { useRouter } from 'next/navigation';

export function useAuth() {
  const [user, setUser] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [isHeadManager, setIsHeadManager] = useState(false);
  const router = useRouter();

  useEffect(() => {
    const unsubscribe = auth.onAuthStateChanged(async (authUser) => {
      if (authUser) {
        // 권한 확인
        const adminDoc = await getDoc(doc(db, 'admins', authUser.uid));

        if (adminDoc.exists()) {
          const adminData = adminDoc.data();

          if (adminData.role === 'HEAD_MANAGER' || adminData.role === 'SUPER_ADMIN') {
            setUser({ ...authUser, ...adminData });
            setIsHeadManager(true);
          } else {
            alert('총관리자 권한이 없습니다.');
            await auth.signOut();
            router.push('/login');
          }
        } else {
          alert('관리자 정보를 찾을 수 없습니다.');
          await auth.signOut();
          router.push('/login');
        }
      } else {
        router.push('/login');
      }

      setLoading(false);
    });

    return () => unsubscribe();
  }, [router]);

  return { user, loading, isHeadManager };
}
```

### 2.6 대시보드 업데이트 (Firestore 연결 테스트)

```tsx
// app/dashboard/page.tsx
'use client';

import { useEffect, useState } from 'react';
import { useAuth } from '@/lib/hooks/useAuth';
import { db } from '@/lib/firebase';
import { collection, getDocs, query, limit } from 'firebase/firestore';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';

export default function DashboardPage() {
  const { user, loading } = useAuth();
  const [officesCount, setOfficesCount] = useState(0);
  const [dataLoading, setDataLoading] = useState(true);

  useEffect(() => {
    async function fetchData() {
      try {
        // 사무실 개수 조회 테스트
        const officesRef = collection(db, 'regions/Hongchon/offices');
        const snapshot = await getDocs(query(officesRef, limit(100)));

        setOfficesCount(snapshot.size);
      } catch (error) {
        console.error('데이터 조회 오류:', error);
      } finally {
        setDataLoading(false);
      }
    }

    if (user) {
      fetchData();
    }
  }, [user]);

  if (loading || dataLoading) {
    return (
      <div className="flex h-screen items-center justify-center">
        <div className="text-lg">로딩 중...</div>
      </div>
    );
  }

  return (
    <div className="p-6 space-y-6">
      <div>
        <h1 className="text-3xl font-bold">대시보드</h1>
        <p className="text-gray-600 mt-1">
          환영합니다, {user?.name || user?.email}님
        </p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <Card>
          <CardHeader>
            <CardTitle className="text-sm font-medium text-gray-600">
              총 사무실
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="text-3xl font-bold">{officesCount}개</div>
          </CardContent>
        </Card>

        <Card>
          <CardHeader>
            <CardTitle className="text-sm font-medium text-gray-600">
              Firestore 연결
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="text-3xl font-bold text-green-600">✓</div>
            <p className="text-sm text-gray-600 mt-1">연결됨</p>
          </CardContent>
        </Card>

        <Card>
          <CardHeader>
            <CardTitle className="text-sm font-medium text-gray-600">
              권한
            </CardTitle>
          </CardHeader>
          <CardContent>
            <div className="text-lg font-bold">{user?.role}</div>
            <p className="text-sm text-gray-600 mt-1">총관리자</p>
          </CardContent>
        </Card>
      </div>

      <Card>
        <CardHeader>
          <CardTitle>Phase 2 완료</CardTitle>
        </CardHeader>
```
