### 3. Repository 패턴 (2개)

#### `CallRepository.kt`
**위치**: `call_manager/app/src/main/java/com/designated/callmanager/data/repository/`

**핵심 패턴**: Local-First + Firebase Background
```kotlin
class CallRepository(
    private val database: AppDatabase,
    private val firestore: FirebaseFirestore,
    private val scope: CoroutineScope
) {
    // ✅ 로컬 우선 읽기
    fun getCallsFlow(regionId: String, officeId: String): Flow<List<CallInfo>> {
        return callDao.getCallsFlow(regionId, officeId)
            .map { localCalls -> localCalls.map { it.toFirebaseCallInfo() } }
    }

    // ✅ 낙관적 업데이트
    suspend fun createCall(callInfo: CallInfo): String {
        // 1. 로컬 즉시 저장 (UI 즉시 반영)
        val tempId = "temp_${System.currentTimeMillis()}"
        callDao.insertCall(LocalCallInfo.fromFirebaseCallInfo(callInfo.copy(id = tempId)))

        // 2. Firebase 백그라운드 업로드
        scope.launch {
            val docRef = firestore.collection("calls").add(callInfo).await()
            callDao.deleteCall(tempId)
            callDao.insertCall(localCall.copy(id = docRef.id, synced = true))
        }

        return tempId
    }

    // ✅ 배차도 동일 패턴
    suspend fun assignCall(callId: String, driverId: String, ...) {
        // 1. 로컬 즉시 업데이트
        callDao.updateCallStatus(callId, "ASSIGNED", driverId)

        // 2. Firebase 백그라운드 업데이트
        scope.launch {
            firestore.collection("calls").document(callId).update(...).await()
        }
    }

    // ✅ 최소한의 리스너 (최근 10개만)
    private fun startSync(regionId: String, officeId: String) {
        syncListener = firestore.collection("calls")
            .orderBy("timestamp", Query.Direction.DESCENDING)
            .limit(10)  // 기존 100 → 10
            .addSnapshotListener { ... }
    }
}
```

- `assignCall()` - 낙관적 업데이트
- `updateCallStatus()` - 낙관적 업데이트
