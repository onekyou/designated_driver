  // 새로운 토큰 기반 매칭 함수
  export const matchByToken = onCall(async (request) => {
    const { token } = request.data

    const tokenDoc = await db.collection('attributionTokens')
      .document(token)
      .get()

    if (!tokenDoc.exists) {
      return { success: false, message: '유효하지 않은 토큰' }
    }

    const data = tokenDoc.data()

    // 만료 확인
    if (data.expiresAt.toMillis() < Date.now()) {
      return { success: false, message: '만료된 QR 코드' }
    }

    // 이미 사용됨 확인
    if (data.status === 'claimed') {
      return { success: false, message: '이미 사용된 QR 코드' }
    }

    return {
      success: true,
      regionId: data.regionId,
      officeId: data.officeId,
      officePhone: data.officePhone,
      bankName: data.bankName,
      accountNumber: data.accountNumber,
      accountHolder: data.accountHolder
    }
  })

  // 토큰 생성 함수 추가
  fun generateAttributionToken(
      regionId: String,
      officeId: String,
      phone: String,
      bankName: String,
      accountNumber: String,
      accountHolder: String
  ): String {
      val token = java.util.UUID.randomUUID().toString()

      // Firestore에 토큰 등록
      val tokenData = mapOf(
          "token" to token,
          "officeId" to officeId,
          "regionId" to regionId,
          "officePhone" to phone,
          "bankName" to bankName,
          "accountNumber" to accountNumber,
          "accountHolder" to accountHolder,
          "createdAt" to com.google.firebase.Timestamp.now(),
          "expiresAt" to com.google.firebase.Timestamp.fromMillis(
              System.currentTimeMillis() + 7 * 24 * 60 * 60 * 1000
          ),
          "status" to "pending"
      )

      firestore.collection("attributionTokens")
          .document(token)
          .set(tokenData)
          .addOnSuccessListener {
              android.util.Log.d(TAG, "토큰 생성 완료: $token")
          }

      return token
  }

  export default function LandingPage() {
    const [token, setToken] = useState<string>('')
    const [hasError, setHasError] = useState(false)

    useEffect(() => {
      const params = new URLSearchParams(window.location.search)
      const t = params.get('token')

      if (t) {
        // 새 방식: 토큰 기반
        setToken(t)
        handleTokenFlow(t)
      } else {
        // 기존 방식: 핑거프린트 (하위 호환)
        const r = params.get('r')
        const o = params.get('o')
        if (r && o) {
          handleLegacyFlow(r, o, ...)
        } else {
          setHasError(true)
        }
      }
    }, [])

    const handleTokenFlow = async (token: string) => {
      // 토큰 유효성 확인 (선택적)
      const tokenDoc = await getDoc(doc(db, 'attributionTokens', token))
      if (!tokenDoc.exists()) {
        setHasError(true)
        return
      }

      // 다운로드 페이지로 리다이렉트 (토큰 포함)
      setTimeout(() => {
        window.location.href = `/download?token=${token}`
      }, 1000)
    }
  }

  export default function DownloadPage() {
    const [token, setToken] = useState<string>('')

    useEffect(() => {
      const params = new URLSearchParams(window.location.search)
      setToken(params.get('token') || '')
    }, [])

    const handleDownload = () => {
      // APK 다운로드 with token
      const apkUrl = '/customer_app.apk'

      // Android Install Referrer 설정
      if (token) {
        const referrer = `token=${token}`
        window.location.href = `${apkUrl}#Intent;S.referrer=${encodeURIComponent(referrer)};end`
      } else {
        // 폴백: 일반 다운로드
        window.location.href = apkUrl
      }
    }
  }

  suspend fun matchByToken(token: String): OfficeInfo? {
      val functions = FirebaseFunctions.getInstance("asia-northeast3")
      val result = functions.getHttpsCallable("matchByToken")
          .call(mapOf("token" to token))
          .await()

      val data = result.data as? Map<*, *>
