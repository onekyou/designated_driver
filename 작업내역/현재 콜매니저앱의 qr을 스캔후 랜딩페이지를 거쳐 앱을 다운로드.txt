  fun clearUserData() {
      prefs.edit().apply {
          remove(KEY_PHONE_NUMBER)
          remove(KEY_IS_PHONE_VERIFIED)
          remove(KEY_CUSTOMER_GRADE)
          remove(KEY_HOME_ADDRESS)
          remove(KEY_FAVORITE_ADDRESSES)
          remove(KEY_OFFICE_ID)        // ← 추가
          remove(KEY_REGION_ID)        // ← 추가
          remove(KEY_OFFICE_PHONE)     // ← 추가
          remove(KEY_BANK_NAME)        // ← 추가
          remove(KEY_ACCOUNT_NUMBER)   // ← 추가
          remove(KEY_ACCOUNT_HOLDER)   // ← 추가
          apply()
      }
  }
