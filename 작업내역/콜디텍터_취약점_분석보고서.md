// 새 파일: utils/PhoneNumberUtils.kt
object PhoneNumberUtils {
    /**
     * 전화번호 정규화
     * 010-1234-5678 → 01012345678
     * +82-10-1234-5678 → 01012345678
     */
    fun normalize(phoneNumber: String): String {
        var normalized = phoneNumber.replace(Regex("[^0-9]"), "")

        // +82 또는 82로 시작하는 국제번호 처리
        if (normalized.startsWith("82") && normalized.length >= 11) {
            normalized = "0" + normalized.substring(2)
        }

        return normalized
    }

    /**
     * 전화번호 포맷팅 (표시용)
     * 01012345678 → 010-1234-5678
     */
    fun format(phoneNumber: String): String {
        return when (phoneNumber.length) {
            11 -> "${phoneNumber.substring(0, 3)}-${phoneNumber.substring(3, 7)}-${phoneNumber.substring(7)}"
            10 -> if (phoneNumber.startsWith("02")) {
                "${phoneNumber.substring(0, 2)}-${phoneNumber.substring(2, 6)}-${phoneNumber.substring(6)}"
            } else {
                "${phoneNumber.substring(0, 3)}-${phoneNumber.substring(3, 6)}-${phoneNumber.substring(6)}"
            }
            else -> phoneNumber
        }
    }
}

// 사용
val normalized = PhoneNumberUtils.normalize(phoneNumber)
val formatted = PhoneNumberUtils.format(normalized)
```

// 새 파일: data/CallStateManager.kt
object CallStateManager {
    private val lock = Any()

    @Volatile private var currentPhoneNumber: String? = null
    @Volatile private var currentCallState: Int = TelephonyManager.CALL_STATE_IDLE
    @Volatile private var lastStateChangeTime: Long = 0

    private const val DUPLICATE_THRESHOLD_MS = 5000L

    fun updateCallState(
        phoneNumber: String?,
        state: Int,
        isIncoming: Boolean
    ): CallStateUpdate {
        synchronized(lock) {
            val now = System.currentTimeMillis()
            val isDuplicate = (phoneNumber == currentPhoneNumber) &&
                              (now - lastStateChangeTime) < DUPLICATE_THRESHOLD_MS

            if (!isDuplicate) {
                currentPhoneNumber = phoneNumber
                currentCallState = state
                lastStateChangeTime = now
            }

            return CallStateUpdate(
                phoneNumber = phoneNumber,
                state = state,
                isIncoming = isIncoming,
                isDuplicate = isDuplicate,
                timestamp = now
            )
        }
    }

    fun reset() {
        synchronized(lock) {
            currentPhoneNumber = null
            currentCallState = TelephonyManager.CALL_STATE_IDLE
        }
    }
}

data class CallStateUpdate(
    val phoneNumber: String?,
    val state: Int,
    val isIncoming: Boolean,
    val isDuplicate: Boolean,
    val timestamp: Long
)
```

### 8. 부재중 전화 미감지 ⭐⭐

**심각도**: MEDIUM
**기능 누락**: 중요 콜 손실 가능

#### 문제 설명
RINGING → IDLE 흐름(전화 받지 않고 끊음)을 처리하지 않음

#### 발견 위치
`CallDetectorService.kt:115-158`
```kotlin
// IDLE 상태 처리
if (callState == TelephonyManager.CALL_STATE_IDLE) {
    if (phoneNumber == lastProcessedPhoneNumber) {
        // ✅ OFFHOOK → IDLE (통화 후 종료)만 처리
        // ❌ RINGING → IDLE (부재중) 미처리
    }
}
```

#### 시나리오
```
1. 전화가 옴 (RINGING)
2. 사용자가 받지 않음
3. 전화가 끊김 (IDLE)
4. ❌ Firebase에 저장 안 됨
5. ❌ 콜매니저에 알림 안 감
```

#### 해결 방안
```kotlin
// CallDetectorService.kt
private var wasRinging = false

override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
    val callState = intent?.getIntExtra("EXTRA_CALL_STATE", TelephonyManager.CALL_STATE_IDLE)
    val phoneNumber = intent?.getStringExtra("incomingPhoneNumber")

    when (callState) {
        TelephonyManager.CALL_STATE_RINGING -> {
            lastProcessedPhoneNumber = phoneNumber
            wasRinging = true
            // RINGING 상태에서 마감 체크는 그대로 유지
        }

        TelephonyManager.CALL_STATE_OFFHOOK -> {
            wasRinging = false  // 받았으므로 부재중 아님
        }

        TelephonyManager.CALL_STATE_IDLE -> {
            if (phoneNumber == lastProcessedPhoneNumber) {
                if (wasRinging) {
                    // 부재중 전화 처리
                    Log.i(TAG, "📞 부재중 전화 감지: $phoneNumber")
                    handleMissedCall(phoneNumber)
                } else {
                    // 통화 종료 처리 (기존 로직)
                    handleCallEnded(phoneNumber)
                }
            }
            wasRinging = false
            lastProcessedPhoneNumber = null
        }
    }

    return START_NOT_STICKY
}

private fun handleMissedCall(phoneNumber: String) {
    serviceScope.launch {
        val regionId = sharedPreferences.getString("regionId", null)
        val officeId = sharedPreferences.getString("officeId", null)

        if (regionId != null && officeId != null) {
            checkOfficeStatusForMissedCall(regionId, officeId, phoneNumber, null)
        }
    }
}
```

private val backupScope = CoroutineScope(Dispatchers.IO)

fun performAutoBackup(onComplete: ((Boolean) -> Unit)? = null) {
    backupScope.launch {
        try {
            val success = backupToFile()
            withContext(Dispatchers.Main) {
                onComplete?.invoke(success)
            }
        } catch (e: Exception) {
            Log.e(TAG, "자동 백업 실패", e)
            withContext(Dispatchers.Main) {
                onComplete?.invoke(false)
            }
        }
    }
}

class BackupWorker(context: Context, params: WorkerParameters) : Worker(context, params) {
    override fun doWork(): Result {
        val manager = ExcludeNumberManager(applicationContext)
        return if (manager.backupToFile()) {
            Result.success()
        } else {
            Result.failure()
        }
    }
}
