import io.agora.rtm.RtmClientListener
import io.agora.rtm.RtmMessage
import io.agora.rtm.RtmResult
import io.agora.rtm.RtmConstants
// ... 기타 필요한 import

class RTMManager(private val rtmClient: RtmClient) : RtmClientListener {

    // RTM 초기화 및 로그인 예시
    fun loginRTM(token: String, userId: String, callback: (RtmResult<Unit>) -> Unit) {
        rtmClient.login(token, userId) { response ->
            if (response.isSuccess) {
                callback(RtmResult.success(Unit)) // RtmResult.success(Unit) 사용
            } else {
                callback(RtmResult.failure(Exception("Login failed: ${response.error?.errorDescription}"))) // RtmResult.failure 사용
            }
        }
    }

    // 메시지 전송 예시 (RtmMessage 대신 String을 사용하거나, RtmMessage 생성자에 따라 수정)
    fun sendMessage(channelName: String, message: String, callback: (RtmResult<Unit>) -> Unit) {
        val channel = rtmClient.createStreamChannel(channelName) // 또는 getChannel(channelName)
        channel?.let {
            it.join { joinResponse ->
                if (joinResponse.isSuccess) {
                    val rtmMessage = RtmMessage.Builder()
                        .text(message)
                        .build()

                    it.sendMessage(rtmMessage) { sendResponse ->
                        if (sendResponse.isSuccess) {
                            callback(RtmResult.success(Unit))
                        } else {
                            callback(RtmResult.failure(Exception("Send message failed: ${sendResponse.error?.errorDescription}")))
                        }
                    }
                } else {
                    callback(RtmResult.failure(Exception("Join channel failed: ${joinResponse.error?.errorDescription}")))
                }
            }
        } ?: callback(RtmResult.failure(Exception("Channel $channelName not found or created")))
    }


    // RtmClientListener 구현
    override fun onMessageReceived(message: RtmMessage, peerId: String) {
        // 메시지 수신 처리
        println("Message received from $peerId: ${message.text}")
    }

    override fun onConnectionStateChanged(state: RtmConstants.RtmConnectionState, reason: RtmConstants.RtmConnectionChangeReason) {
        // 연결 상태 변경 처리
        println("Connection state changed: $state, reason: $reason")
    }

    // ... 기타 필요한 RtmClientListener 오버라이드
}
  Unresolved reference 오류:많은 Unresolved reference 오류는 주로 다음 두 가지 원인으로 발생합니다.올바른 import 문 누락: RTM 2.x API의 클래스나 함수는 특정 패키지에 속해 있습니다. 사용하는 클래스에 대한 import 문이 코드 상단에 정확히 추가되었는지 확인하세요.예: import io.agora.rtm.RtmClient, import io.agora.rtm.RtmMessage, import io.agora.rtm.RtmConstants 등API 변경: RTM 1.x에서 2.x로 넘어오면서 일부 API 이름, 파라미터, 또는 반환 타입이 변경되었을 수 있습니다. Agora RTM 2.x 공식 문서를 참조하여 사용하려는 API의 정확한 시그니처를 확인해야 합니다.resultCallback: RTM 2.x에서는 ResultCallback 대신 람다((RtmResult<Unit>) -> Unit)나 RtmEventListener를 사용하는 경우가 많습니다.messageType: RtmMessage 객체를 생성할 때 messageType 대신 text() 또는 다른 빌더 메서드를 사용할 수 있습니다.RtmConstants.RtmPresenceEventType, RtmConstants.RtmPresenceEvent: Presence 기능 관련 API도 2.x에서 변경되었을 수 있습니다.특정 오류 메시지 해결 가이드:No value passed for parameter 'resultCallback': RTM 2.x의 비동기 메서드는 콜백을 람다 함수 형태로 받거나, 코루틴의 suspend 함수로 구현되어 있을 수 있습니다. 해당 메서드 호출 시 올바른 콜백 파라미터를 제공해야 합니다.Unresolved reference: errorinfo: errorinfo는 더 이상 직접 접근하는 속성이 아닐 수 있습니다. RtmResult 객체의 error 속성 내에 오류 설명이 포함되어 있을 가능성이 높습니다.Unresolved reference: publishTextMassage: publishTextMassage 대신 sendMessage와 같은 API를 사용해야 합니다.Unresolved reference: RtmConnectionState, RtmConnectionChangeReason, messageType: 이들은 io.agora.rtm.RtmConstants 또는 io.agora.rtm.RtmMessage 클래스 내에 정의되어 있으므로 올바른 import와 접근 방식을 사용해야 합니다.None of the following functions can be called with the arguments supplied...: 이는 함수 오버로드 문제로, 인자 타입이나 개수가 일치하지 않을 때 발생합니다. 사용하는 함수의 정확한 시그니처를 확인하세요.Cannot access 'publisher': it is private in 'PresenceEvent': PresenceEvent의 publisher 필드가 private으로 선언되어 있어 직접 접근할 수 없다는 의미입니다. 해당 값을 가져오는 공용 메서드(getter)가 있는지 확인하거나, Presence Event 처리 방식을 검토해야 합니다.3. Agora RTM 2.x 공식 문서 참조가장 정확한 정보는 Agora RTM 2.x Android SDK의 공식 문서입니다. 사용하려는 모든 API의 정확한 시그니처, 클래스, 그리고 사용법을 여기서 확인할 수 있습니다.요약 및 재시도 방법:build.gradle 파일의 RTM SDK 버전 확인 및 업데이트: 최신 RTM 2.x 버전을 사용하고 있는지 확인합니다.import 문 검토: 모든 RTM 관련 클래스와 함수에 대한 import 문이 정확히 추가되었는지 확인합니다. Agora RTM 2.x 공식 문서를 참조하여 사용하려는 API (예: 로그인, 메시지 전송, 채널 가입 등)의 정확한 파라미터와 반환 타입을 확인하고 코드를 수정합니다.콜백 및 비동기 처리 방식 변경: Result<Unit> 또는 RtmResult를 사용하는 방식으로 비동기 콜백을 처리하도록 코드를 변경합니다.이러한 단계를 따르면 대부분의 오류를 해결할 수 있을 것입니다. 특정 오류에 대해 더 자세한 도움이 필요하면 해당 코드 스니펫과 함께 다시 문의해주세요.
