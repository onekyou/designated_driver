class PTTAccessibilityService : AccessibilityService() {
    private var soundPool: SoundPool? = null
    private var beepSoundId: Int = 0
    private var isSoundLoaded = false

    override fun onCreate() {
        super.onCreate()
        initializeSoundPool()
    }

    override fun onServiceConnected() {
        super.onServiceConnected()
        Log.d(TAG, "Accessibility Service Connected")
        // Consider moving sound pool initialization here if context is an issue
    }

    private fun initializeSoundPool() {
        val audioAttributes = AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_MEDIA)
            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
            .build()

        soundPool = SoundPool.Builder()
            .setMaxStreams(1) // Tried with higher numbers too, no change
            .setAudioAttributes(audioAttributes)
            .build()

        soundPool?.setOnLoadCompleteListener { _, sampleId, status ->
            if (status == 0) {
                isSoundLoaded = true
                Log.d(TAG, "Sound loaded successfully. Sound ID: $sampleId")
            } else {
                Log.e(TAG, "Sound failed to load. Status: $status")
            }
        }

        beepSoundId = soundPool?.load(this, R.raw.ptt_beep, 1) ?: 0
        if (beepSoundId == 0) {
            Log.e(TAG, "Failed to load sound resource R.raw.ptt_beep")
        }
    }

    private fun playImmediateBeep() {
        if (soundPool != null && beepSoundId != 0 && isSoundLoaded) {
            val streamId = soundPool!!.play(beepSoundId, 1.0f, 1.0f, 1, 0, 1.0f)
            Log.d(TAG, "Attempted to play sound. Stream ID: $streamId")
            if (streamId == 0) {
                Log.w(TAG, "SoundPool.play() returned 0, indicating failure or no available streams.")
            }
        } else {
            Log.w(TAG, "SoundPool not ready: soundPool=$soundPool, beepSoundId=$beepSoundId, isSoundLoaded=$isSoundLoaded")
        }
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        // Assume playImmediateBeep() is called here based on volume key events
        // e.g., if (event?.eventType == AccessibilityEvent.TYPE_KEY_EVENT && KeyEvent.ACTION_DOWN) {
        //     playImmediateBeep()
        // }
    }

    override fun onInterrupt() {
        Log.w(TAG, "Accessibility service interrupted.")
    }

    override fun onDestroy() {
        super.onDestroy()
        soundPool?.release()
        soundPool = null
        Log.d(TAG, "SoundPool released and service destroyed.")
    }

    companion object {
        private const val TAG = "PTTAccessibilityService"
    }
}
