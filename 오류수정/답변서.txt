import android.util.Log
import io.agora.rtm.LockEvent
import io.agora.rtm.MessageEvent
import io.agora.rtm.PresenceEvent
import io.agora.rtm.RtmClient
import io.agora.rtm.RtmClient.RtmConfig
import io.agora.rtm.RtmEventListener
import io.agora.rtm.StorageEvent
import io.agora.rtm.TopicEvent
import io.agora.rtm.internal.RtmConstants.RtmMessageOperation
import org.json.JSONException
import org.json.JSONObject

class AgoraRtmManager(private val context: Context, private val appId: String, private val userId: String) {

    private val TAG = "AgoraRtmManager"
    private var rtmClient: RtmClient? = null

    // RtmEventListener 구현
    private val rtmEventListener = object : RtmEventListener {
        override fun onMessageEvent(event: MessageEvent) {
            val message = event.message
            val publisher = event.publisher // 메시지 발신자 ID
            val channelName = event.channelName // 메시지가 수신된 채널 이름

            Log.d(TAG, "Message received from $publisher in channel $channelName")
            Log.d(TAG, "Message type: ${message.messageType}")

            try {
                // message.payload는 ByteArray이므로 String으로 변환
                val messageContent = String(message.payload, Charsets.UTF_8)
                Log.d(TAG, "Message content: $messageContent")

                // JSON 문자열 파싱 예시
                try {
                    val jsonObject = JSONObject(messageContent)
                    val customData = jsonObject.optString("data", "No custom data")
                    Log.d(TAG, "Parsed JSON data: $customData")
                    // TODO: UI 업데이트 또는 PTT 로직 처리
                } catch (e: JSONException) {
                    Log.e(TAG, "Failed to parse message as JSON: $e")
                    // JSON이 아닌 일반 문자열 메시지 처리
                }

            } catch (e: Exception) {
                Log.e(TAG, "Error processing message payload: $e")
            }
        }

        override fun onPresenceEvent(event: PresenceEvent) {
            Log.d(TAG, "Presence event received in channel ${event.channelName}")
            Log.d(TAG, "Publisher: ${event.publisher}, Type: ${event.type}")

            when (event.type) {
                RtmConstants.RtmPresenceEventType.REMOTE_JOIN -> {
                    Log.d(TAG, "${event.publisher} joined channel ${event.channelName}")
                }
                RtmConstants.RtmPresenceEventType.REMOTE_LEAVE -> {
                    Log.d(TAG, "${event.publisher} left channel ${event.channelName}")
                }
                RtmConstants.RtmPresenceEventType.REMOTE_TIMEOUT -> {
                    Log.d(TAG, "${event.publisher} timed out in channel ${event.channelName}")
                }
                else -> {
                    // 기타 Presence 이벤트 처리
                }
            }
        }

        override fun onTopicEvent(event: TopicEvent) {
            Log.d(TAG, "Topic event received: ${event.topicName}")
        }

        override fun onLockEvent(event: LockEvent) {
            Log.d(TAG, "Lock event received: ${event.channelName}, ${event.lockName}, ${event.eventType}")
        }

        override fun onStorageEvent(event: StorageEvent) {
            Log.d(TAG, "Storage event received: ${event.eventType}, ${event.target}")
            when (event.eventType) {
                RtmConstants.RtmStorageEventType.USER_METADATA_UPDATED -> {
                    Log.d(TAG, "User metadata updated for ${event.target}")
                }
                RtmConstants.RtmStorageEventType.CHANNEL_METADATA_UPDATED -> {
                    Log.d(TAG, "Channel metadata updated for ${event.target}")
                }
                else -> {
                    // 기타 Storage 이벤트 처리
                }
            }
        }
    }

    fun initAndLogin() {
        val config = RtmConfig.Builder(appId, userId)
            .setRtmEventListener(rtmEventListener) // RTM 2.x에서 이벤트 리스너 설정 방식
            .build()

        RtmClient.create(config, object : io.agora.rtm.ResultCallback<RtmClient> {
            override fun onSuccess(client: RtmClient?) {
                rtmClient = client
                Log.d(TAG, "RTM Client created successfully.")
                login()
            }

            override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                Log.e(TAG, "Failed to create RTM Client: ${error?.errorReason}")
            }
        })
    }

    private fun login() {
        rtmClient?.login(object : io.agora.rtm.ResultCallback<Void> {
            override fun onSuccess(response: Void?) {
                Log.d(TAG, "RTM logged in successfully.")
                // 로그인 성공 후 채널 구독 등 추가 작업 수행
            }

            override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                Log.e(TAG, "Failed to login to RTM: ${error?.errorReason}")
            }
        })
    }

    fun logout() {
        rtmClient?.logout(object : io.agora.rtm.ResultCallback<Void> {
            override fun onSuccess(response: Void?) {
                Log.d(TAG, "RTM logged out successfully.")
            }

            override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                Log.e(TAG, "Failed to logout from RTM: ${error?.errorReason}")
            }
        })
    }

    fun joinChannel(channelName: String) {
        val options = io.agora.rtm.ChannelSubscribeOptions.Builder().build()
        rtmClient?.getChannel(channelName)?.subscribe(options, object : io.agora.rtm.ResultCallback<Void> {
            override fun onSuccess(response: Void?) {
                Log.d(TAG, "Joined channel $channelName successfully.")
            }

            override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                Log.e(TAG, "Failed to join channel $channelName: ${error?.errorReason}")
            }
        })
    }

    fun leaveChannel(channelName: String) {
        rtmClient?.getChannel(channelName)?.unsubscribe(object : io.agora.rtm.ResultCallback<Void> {
            override fun onSuccess(response: Void?) {
                Log.d(TAG, "Left channel $channelName successfully.")
            }

            override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                Log.e(TAG, "Failed to leave channel $channelName: ${error?.errorReason}")
            }
        })
    }

    fun publishMessage(channelName: String, message: String) {
        val rtmMessage = rtmClient?.createMessage(message.toByteArray(Charsets.UTF_8))
        if (rtmMessage != null) {
            val options = io.agora.rtm.PublishOptions.Builder().build()
            rtmClient?.getChannel(channelName)?.publish(rtmMessage, options, object : io.agora.rtm.ResultCallback<Void> {
                override fun onSuccess(response: Void?) {
                    Log.d(TAG, "Message published to $channelName successfully: $message")
                }

                override fun onFailure(error: io.agora.rtm.ErrorInfo?) {
                    Log.e(TAG, "Failed to publish message to $channelName: ${error?.errorReason}")
                }
            })
        }
    }

    fun release() {
        rtmClient?.release()
        rtmClient = null
        Log.d(TAG, "RTM Client released.")
    }
}
